int main()
{
    //1) 分别定义 char 、int 、float 、double 变量和对应的指针变量，并给他们赋值，最后打印结果
    char c;
    int i;
    float f;
    double d;
    
    char *pc = &c;
    int *pi = &i;
    float *pf = &f;
    double *pd = &d;
    
    cout<<"please input char in float double :";
    cin>>*pc>>*pi>>*pf>>*pd;
    cout<<c<<"\n";
    cout<<i<<"\n";
    cout<<f<<"\n";
    cout<<d<<"\n";

    
    //2) 输入n个数，找其中的中数（从大到小排序后中间的数）
    int j,n,an[256];
    cout<<"please input n(<256):";
    cin>>n;
    int *pj,*pn = an;
    for( i=0;i<n;i++,pn++ ){
        cin>>*pn;
    }
    for( i=0,pn=an;i<n;i++,pn++ ){
        for( pj=pn+1,j=i+1;j<n;j++,pj++ ){
            if ( *pn>*pj ){
                int t = *pn;
                *pn = *pj;
                *pj = t;
            }
        }
    }
    cout<<"The middle num is:"<<an[n/2]<<"\n";
    
    //3) 定义一个3*3 二维数组，输入数据，然后用指针访问并输出矩阵
    int a3x[3][3];
    int *px = (int*)a3x;
    cout<<"please input 3*3: \n";
    for( i=0;i<3;i++ ){
        for( j=0;j<3;j++ ){
            cin>>*px++;
        }
    }
    px = (int*)a3x;
    for( i=0;i<3;i++ ){
        for( j=0;j<3;j++ ){
            cout<< *px++ <<" ";
        }
        cout<<"\n"; 
    }

    // 4) 先输入一个字符串，然后删除其中某个字符（有可能多个），最后统计并输出结果字符串长度
    char ss,s1[256],s2[256];
    cout<<"please intput string:";
    gets_s(s1);
    cout<<"please input delete char: ";
    cin>>ss;
    char *p1 = s1,*p2 = s2;
    for( i=0;i<256;i++,p1++ ){
        if ( *p1==0 ) break; // 字符串结束 
        if ( *p1!=ss ){
            *p2++ = *p1;
        }
    }
    *p2 = 0; // 末尾加0 字符数组升级为字符串
    cout<<s2;
    cout<<"\n";
   
    // 5) 先输入一个字符串，编程实现字符串调头（头尾交换）
    cout<<"please input string:";
    gets_s(s1);
    cout<<"\n";
    int s1sz=0; 
    for( p1=s1,i=0;i<256;i++,p1++ ){ // 获取字符串长度
        if ( *p1==0 ){ // 找到“结束位置”
            s1sz = i;
            break;  
        }
    }
    for( p1=s1,p2=s1+s1sz-1,i=0;i<s1sz/2;i++,p1++,p2-- ){
        char tc = *p1;
        *p1 = *p2;
        *p2 = tc;
    }
    cout<<s1;
    cout<<"\n";
    
    //6) 输入一个长字符串和短字符串，统计短字符串在长字符串中出现的次数（也就是word中的查找功能）
    cout<<"please input long string:";
    gets_s(s1);
    cout<<"please input short string:";
    gets_s(s2);
    int s2sz=0,findSum=0;
    for( p2=s2,i=0;i<256;i++,p2++ ){ // 获取短字符串长度
        if (*p2==0){ 
            s2sz = i; 
            break;
        } 
    }
    for( p1=s1;*p1!=0;p1++ ){
        if (*p1==0) break;
        for( p2=s2,i=0;i<s2sz;i++,p2++ ){
            if ( *p2!=*(p1+i) ) break;
        }
        if ( i==s2sz ) findSum++;
    }
    cout<<"Find sum: "<< findSum;


    system("pause");
    return 0;
}
