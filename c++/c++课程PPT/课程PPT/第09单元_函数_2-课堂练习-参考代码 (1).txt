//1.编写函数实现 ∑n 也就是 0+1+2…+n 的值
int  GetSum( int *a,int n ){
    int sum,i;
    for( sum=0,i=0;i<n;i++,a++ ){
        sum += *a; 
    }
    return sum;
}

//2.分别编写数组求最大、最小、均值、中值的函数
double GetMax(double *x, int n) {
    double mx = x[0];
    for (int i = 0; i < n; i++) {
        if (mx < x[i]) mx = x[i];
    }
    return mx;
}
double GetMin(double *x, int n) {
    double mx = x[0];
    for (int i = 0; i < n; i++) {
        if (mx > x[i]) mx = x[i];
    }
    return mx;
}
double GetAvg(double *x, int n) {
    double mx = 0;
    for (int i = 0; i < n; i++,x++ ) {
        mx += *x;
    }
    return mx/n;
}
void Swap(double *x, double *y) {
    double t = x[0]; // t = *x;
    x[0] = y[0];        // *x = *y;
    y[0] = t;            // *y = t;
}
double GetMid(double *x, int n) {
    for (int i = 0; i < n; i++) {
        for (int j = i + 1; j < n; j++) {
            if (x[i] > x[j]) {
                Swap(x + i, x + j);                
            }
        }
    }
    return x[n / 2];
}

//3.编写函数实现将字符串反向
void Resv(char *s) {
    int n,i;
    for (i = 0; i < 100000; i++) {
        if (s[i] == 0){
            n = i;
            break;
        }
    }
    for (i = 0; i < n / 2; i++) {
        char t = s[i];
        s[i] = s[n - 1 - i];
        s[n - 1 - i] = t;
    }
}

//4.分别编写自己的 mystrlen, mystrcpy，mystrcmp 函数
int mystrlen(const char *s){
    const char *s0=s; while(*s!=0){ s++; }
    return (s-s0);
}
void mystrcpy(char *strDes,const char *strSrc){
    while(*strSrc!=0){ *strDes++ = *strSrc++; }
    *strDes = 0; // 添加结束标志
}
int mystrcmp(const char *s1,const char *s2){
    while ( *s1!=0 && *s1==*s2 ){
        s1++; s2++;
    }
    return (*s1-*s2);
}

//5.编写函数实现在字符串中删除子支付串
bool DelStr( char *strSrc,const char *strSub ){
    bool ret = false;
    int subSz = mystrlen(strSub);
    char *s1,*p1=strSrc,*p2=strSrc;
    const char*s2;  
    while ( *p2!=0 ){        
        for( s1=p2,s2=strSub;*s1==*s2;s1++,s2++ ); //  查找字符串
        if ( *s2==0 ){
            p2 += subSz; 
            ret = true;
        }else{
            *p1++ = *p2++;
        }
    }
    *p1 = 0; // 添加结束标志
    return ret;
}

// 6.编写 用递归实现阶乘 的函数
double Fac( int n){
    return (n==1)?1:(n*Fac(n-1));
}

//7.编写函数实现判断一个点是否在任意多边形内
inline bool Between(double x0, double y0, double x1, double y1, double x, double y){
    double epsilon = 1E-15;
    double dotproduct = (x - x0) * (x1 - x0) + (y - y0)*(y1 - y0);
    if (dotproduct < -epsilon * epsilon) return false;
    double squarednorm = (x1 - x0) * (x1 - x0) + (y1 - y0) * (y1 - y0);
    if (dotproduct - squarednorm > epsilon * epsilon) return false;
    return true;
}
inline bool Intersect(double x0, double y0, double x1, double y1, double x2, double y2, double x3, double y3, double *x, double *y){
    double dx0 = x1 - x0;  double dy0 = y1 - y0;
    double dx2 = x3 - x2;  double dy2 = y3 - y2;
    double tmp1 = dx2 * dy0 - dx0 * dy2;
    if ((tmp1 < 0 ? -tmp1 : tmp1) < 1.0e-34) { // parr or same
        *x = *y = (double)(1.0e32);
        if (x0 == x2 || x0 == x3) *x = x0;
        if (x1 == x2 || x1 == x3) *x = x1;
        if (y0 == y2 || y0 == y3) *y = y0;
        if (y1 == y2 || y1 == y3) *y = y1;
        return (*x != 1.0e32  && *y != 1.0e32) ? true : false;
    }    
    *x = (dx2*(dx0*(y2 - y0) + dy0 * x0) - dx0 * dy2*x2) / tmp1;
    *y = (dy0*(dy2*(x0 - x2) + dx2 * y2) - dx0 * dy2*y0) / tmp1;    
    if (!Between(x0, y0, x1, y1, *x, *y) || !Between(x2, y2, x3, y3, *x, *y)) return false;    
    return true;
}
bool PtInRgn( double *rgnX,double *rgnY,int rgnPtSum,double x,double y){
    if (rgnPtSum<3) return false;
    int  i,crosstime = 0; 
    double tx,ty,minY=rgnY[0],minX=rgnX[0]; 
    for ( i=0;i<rgnPtSum;i++ ){ if (minX<rgnX[i]) minX=rgnX[i]; }    
    minX -= 1E38; // 无穷小 
    for (i=0;i<rgnPtSum;i++){
        if (i==rgnPtSum-1){ tx=rgnX[0];ty=rgnY[0]; }
        else{  tx=rgnX[i+1];ty=rgnY[i+1]; }
        
        if ( Intersect( x,y,minX,minY,rgnX[i],rgnY[i],tx,ty,&tx,&ty ) )
            crosstime ++;
    }
    return ((crosstime%2)==1);
}
