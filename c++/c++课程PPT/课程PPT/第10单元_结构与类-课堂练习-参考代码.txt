// 1.求距离
struct PT2D {
    float x, y;
};
double GetDis(PT2D p1, PT2D p2) {
    return sqrt((p1.x - p2.x)*(p1.x - p2.x) + 
        (p1.y - p2.y)*(p1.y - p2.y));
}
//2. 找最近点
PT2D GetNearest(PT2D *pA, int n, PT2D pt) {
    double miDis = GetDis(pt, pA[0]);
    PT2D ret = pA[0];
    for (int i = 1; i < n; i++) {
        double curDis = GetDis(pt, pA[i]);
        if (miDis > curDis) {
            ret = pA[i];
        }
    }
    return ret;
}

//3.通过学号查找学生
struct STUD{
    char name[32];
    int id;
    float sc[4];
};
int FindById( STUD *pS,int n,int id ){
    for( int i=0;i<n;i++,pS++ ){
        if ( pS->id==id ){
            return i; // 找到就直接返回序号
        }
    }
    return -1; // 没有找到就返回 -1
}

//4. 通过书名查找书
struct BOOK{
    char bkTitle[128];
    char bkAuthor[16];
    float bkPrice;
};

int FindByAnthor( BOOK *pBk,int n,const char *strAuthor ){
    for( int i=0;i<n;i++,pBk++ ){
        if ( strcmp(pBk->bkAuthor,strAuthor)==0 ){ // 字符串比较必须用 strcmp
            return i; // 找到就直接返回序号
        }
    }
    return -1; // 没有找到就返回 -1
}
