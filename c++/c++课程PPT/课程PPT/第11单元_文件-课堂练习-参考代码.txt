//4. 设计控制点结构体
struct XYZ{
    char name[32];
    float x,y,z;
};
//设计函数
XYZ GetNearest( XYZ *pPt,int n,float x,float y){
    int i;
    double dis0,dis; 
    XYZ ret = *pPt;
    dis0 = sqrt( (pPt->x-x)*(pPt->x-x)+(pPt->y-y)*(pPt->y-y) );
    for( i=0;i<n;i++,pPt++ ){
        dis = sqrt( (pPt->x-x)*(pPt->x-x)+(pPt->y-y)*(pPt->y-y) );
        if ( dis<dis0 ){
            dis0 = dis;
            ret = *pPt;
        }
    }
    return ret;
}

void main(){
